import os

\# 1\. Set the path to your test folder on your Mac  
folder\_path \= "/Users/YourName/Documents/BookTest" 

\# 2\. Loop through the files in the directory  
for filename in os.listdir(folder\_path):  
    \# Ignore hidden Mac files like .DS\_Store  
    if filename.startswith('.'):  
        continue  
          
    \# Split the name and the extension  
    name, extension \= os.path.splitext(filename)  
      
    \# 3\. Clean the title: replace underscores/dashes, strip spaces, apply Title Case  
    clean\_title \= name.replace("\_", " ").replace("-", " ").strip().title()  
      
    \# Reattach the extension (keep it lowercase for uniformity)  
    clean\_filename \= clean\_title \+ extension.lower()  
      
    \# 4\. Construct the full file paths  
    old\_file\_path \= os.path.join(folder\_path, filename)  
    new\_file\_path \= os.path.join(folder\_path, clean\_filename)  
      
    \# 5\. Actually rename the file on the hard drive  
    if old\_file\_path \!= new\_file\_path:  
        os.rename(old\_file\_path, new\_file\_path)  
        print(f"Renamed: '{filename}' \-\> '{clean\_filename}'")

print("Cleanup complete\!")

import os  
import pandas as pd

\# A mock list of messy filenames to practice on  
messy\_files \= \[

    "GREAT GATSBY.pdf",

\]

    \# 1\. Separate the name and the extension  
    name, extension \= os.path.splitext(filename)  
      
    \# 2\. Clean the title: replace underscores/hyphens, strip spaces, Title Case

    \# 3\. Standardize the extension (force lowercase)  
    clean\_ext \= extension.lower()  
      
    \# 4\. Store the data as a dictionary record  
    book\_data.append({  
        "Original Filename": filename,  
        "Cleaned Title": clean\_title,  
        "Extension": clean\_ext  
    })

\# Convert the list of dictionaries into a Pandas DataFrame  
df \= pd.DataFrame(book\_data)

\# Print the DataFrame   
print(df)
